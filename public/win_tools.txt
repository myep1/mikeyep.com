# 1) Fast EBCDIC vs ASCII/UTF-8 detector (reads first ~4 KiB)
function ebc {
  param([Parameter(Mandatory=$true)][string]$path)

  $fs = [IO.File]::Open($path, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::ReadWrite)
  try {
    $sz  = [int][Math]::Min(4096, $fs.Length)
    $buf = New-Object byte[] $sz
    [void]$fs.Read($buf, 0, $sz)
  } finally { $fs.Dispose() }

  $ebcSpace = ($buf | Where-Object { $_ -eq 0x40 }).Count   # EBCDIC space (0x40)
  $ascSpace = ($buf | Where-Object { $_ -eq 0x20 }).Count   # ASCII space  (0x20)

  $ebcLetters = ($buf | Where-Object {
    ($_ -ge 0xC1 -and $_ -le 0xC9) -or  # A–I (EBCDIC)
    ($_ -ge 0xD1 -and $_ -le 0xD9) -or  # J–R
    ($_ -ge 0xE2 -and $_ -le 0xE9)      # S–Z
  }).Count

  $ascLetters = ($buf | Where-Object {
    ($_ -ge 0x41 -and $_ -le 0x5A) -or  # A–Z (ASCII)
    ($_ -ge 0x61 -and $_ -le 0x7A)      # a–z
  }).Count

  if (($ebcSpace -gt $ascSpace) -and ($ebcLetters -gt $ascLetters)) {
    'Likely EBCDIC'
  } else {
    'Likely ASCII/UTF-8'
  }
}

# 2) Simple EBCDIC → UTF-8 (no RDW/BDW handling)
function EbcdicToUtf8 {
  [CmdletBinding()]
  param(
    [Parameter(Mandatory = $true)][string]$inPath,
    [Parameter(Mandatory = $true)][string]$outPath,
    [int]$codePage = 37   # 37=IBM037 (US/CA). Others: 500, 1047, 1140.
  )

  $bytes = [IO.File]::ReadAllBytes($inPath)
  $enc   = [System.Text.Encoding]::GetEncoding($codePage)
  $text  = $enc.GetString($bytes)

  [IO.File]::WriteAllText($outPath, $text, [System.Text.Encoding]::UTF8)
}

# 3) Variable-Blocked (VB) EBCDIC with RDW (and optional BDW) → UTF-8
function ConvertEbcdicVB {
  [CmdletBinding()]
  param(
    [Parameter(Mandatory=$true)][string]$InPath,
    [Parameter(Mandatory=$true)][string]$OutPath,
    [int]$CodePage = 1047,   # Common: 1047/1140; US/CA: 37
    [switch]$HasBDW          # Set if blocks have a 4-byte BDW before RDWs
  )

  $enc = [System.Text.Encoding]::GetEncoding($CodePage)
  $fs  = [IO.File]::Open($InPath, [IO.FileMode]::Open, [IO.FileAccess]::Read, [IO.FileShare]::ReadWrite)
  $sw  = New-Object IO.StreamWriter($OutPath, $false, [System.Text.Encoding]::UTF8)

  try {
    if ($HasBDW) {
      while ($true) {
        $bdw = New-Object byte[] 4
        $n = $fs.Read($bdw,0,4)
        if ($n -lt 4) { break }                                # EOF
        $blockLen = (($bdw[0] -shl 8) + $bdw[1])               # includes BDW
        if ($blockLen -lt 4) { break }
        $remaining = $blockLen - 4

        while ($remaining -gt 0) {
          $rdw = New-Object byte[] 4
          if ($fs.Read($rdw,0,4) -lt 4) { break }
          $recLen = (($rdw[0] -shl 8) + $rdw[1])               # includes RDW
          if ($recLen -lt 4 -or $recLen -gt $remaining) { break }
          $dataLen = $recLen - 4

          $buf = New-Object byte[] $dataLen
          if ($fs.Read($buf,0,$dataLen) -lt $dataLen) { break }

          $line = $enc.GetString($buf).TrimEnd()
          $sw.WriteLine($line)
          $remaining -= $recLen
        }
      }
    } else {
      while ($true) {
        $rdw = New-Object byte[] 4
        $n = $fs.Read($rdw,0,4)
        if ($n -lt 4) { break }                                # EOF
        $recLen = (($rdw[0] -shl 8) + $rdw[1])                 # includes RDW
        if ($recLen -lt 4) { break }
        $dataLen = $recLen - 4

        $buf = New-Object byte[] $dataLen
        if ($fs.Read($buf,0,$dataLen) -lt $dataLen) { break }

        $line = $enc.GetString($buf).TrimEnd()
        $sw.WriteLine($line)
      }
    }
  }
  finally {
    $sw.Dispose()
    $fs.Dispose()
  }
}

# Handy wrapper: cvb IN OUT [codePage] [-bdw]
function cvb {
  param(
    [Parameter(Mandatory = $true)][string]$in,
    [Parameter(Mandatory = $true)][string]$out,
    [int]$cp = 1047,
    [switch]$bdw
  )
  ConvertEbcdicVB -InPath $in -OutPath $out -CodePage $cp -HasBDW:$bdw
}

# ebc .\file.dat
# EbcdicToUtf8 -inPath .\in.dat -outPath .\out.txt -codePage 37
# ConvertEbcdicVB -InPath .\in.dat -OutPath .\out.txt -CodePage 1140 -HasBDW
# cvb .\in.dat .\out.txt            # defaults cp=1047
# cvb .\in.dat .\out.txt 1140 -bdw  # BDW + RDW

# Small hex dump (offset, 16 bytes/row, ASCII at right)
function HexDump {
  [CmdletBinding()]
  param(
    [Parameter(Mandatory=$true)][string]$Path,
    [int64]$Offset = 0,
    [int]$Count = 256
  )

  $fs = [IO.File]::Open($Path,'Open','Read','ReadWrite')
  try {
    [void]$fs.Seek($Offset, 'Begin')
    $buf = New-Object byte[] $Count
    $n = $fs.Read($buf,0,$Count)
    for ($i=0; $i -lt $n; $i+=16) {
      $slice = $buf[$i..([Math]::Min($i+15,$n-1))]
      $hex = ($slice | ForEach-Object { $_.ToString('X2') }) -join ' '
      $asc = -join ($slice | ForEach-Object { if ($_ -ge 32 -and $_ -le 126) {[char]$_} else {'.'} })
      '{0:X8}  {1,-48}  {2}' -f ($Offset + $i), $hex, $asc
    }
  }
  finally { $fs.Dispose() }
}

# Guess common EBCDIC code page (plain or VB via -VB); prints scores
function GuessEbcdicCodePage {
  [CmdletBinding()]
  param(
    [Parameter(Mandatory=$true)][string]$Path,
    [switch]$VB,            # set for RDW VB files (no BDW)
    [int]$MaxBytes = 32768  # sample size
  )

  $bytes = New-Object System.Collections.Generic.List[byte]

  $fs = [IO.File]::Open($Path,'Open','Read','ReadWrite')
  try {
    if ($VB) {
      $read = 0
      while ($read -lt $MaxBytes -and $fs.Position -lt $fs.Length) {
        $rdw = New-Object byte[] 4
        if ($fs.Read($rdw,0,4) -lt 4) { break }
        $recLen = (($rdw[0] -shl 8) + $rdw[1])
        if ($rdw[2] -ne 0 -or $rdw[3] -ne 0 -or $recLen -lt 4) { break }
        $dataLen = [Math]::Min($recLen - 4, $MaxBytes - $read)
        $buf = New-Object byte[] $dataLen
        if ($fs.Read($buf,0,$dataLen) -lt $dataLen) { break }
        $bytes.AddRange($buf)
        $read += $dataLen
      }
    } else {
      $take = [int][Math]::Min($MaxBytes, $fs.Length)
      $buf  = New-Object byte[] $take
      [void]$fs.Read($buf,0,$take)
      $bytes.AddRange($buf)
    }
  } finally { $fs.Dispose() }

  $cps = @(37, 500, 1047, 1140, 1141)
  foreach ($cp in $cps) {
    $enc = [Text.Encoding]::GetEncoding($cp)
    $s   = $enc.GetString($bytes.ToArray())

    $printable = ($s.ToCharArray() | Where-Object { [int]$_ -ge 32 -and [int]$_ -le 126 }).Count
    $ratio = [Math]::Round(($printable / [Math]::Max(1,$s.Length)) * 100, 1)

    $vowels = ([regex]::Matches($s, '[AEIOUaeiou]').Count)
    "{0}: printable={1}% vowels={2}" -f $cp, $ratio, $vowels
  }
}

# Convert Fixed-Blocked (FB) EBCDIC to UTF-8
function ConvertEbcdicFB {
  [CmdletBinding()]
  param(
    [Parameter(Mandatory=$true)][string]$InPath,
    [Parameter(Mandatory=$true)][string]$OutPath,
    [Parameter(Mandatory=$true)][int]$RecordLen,  # LRECL (e.g., 80, 133)
    [int]$CodePage = 1047,
    [switch]$Trim
  )

  $enc = [Text.Encoding]::GetEncoding($CodePage)
  $fs  = [IO.File]::Open($InPath, 'Open', 'Read', 'ReadWrite')
  $sw  = New-Object IO.StreamWriter($OutPath, $false, [Text.Encoding]::UTF8)

  try {
    $buf = New-Object byte[] $RecordLen
    while ($true) {
      $n = $fs.Read($buf,0,$RecordLen)
      if ($n -eq 0) { break }
      if ($n -lt $RecordLen) { $buf = $buf[0..($n-1)] }

      $line = $enc.GetString($buf)
      if ($Trim) { $line = $line.TrimEnd() }
      $sw.WriteLine($line)
      if ($n -lt $RecordLen) { break }
    }
  }
  finally { $sw.Dispose(); $fs.Dispose() }
}

# Detect mainframe dataset style: VB (RDW/BDW) vs FB/plain
function DetectMfFormat {
  [CmdletBinding()]
  param([Parameter(Mandatory=$true)][string]$Path)

  $fs = [IO.File]::Open($Path, 'Open', 'Read', 'ReadWrite')
  try {
    if ($fs.Length -lt 4) { return 'Too small / unknown' }

    $h = New-Object byte[] 4
    [void]$fs.Read($h,0,4)
    $len1 = ($h[0] -shl 8) + $h[1]
    $rdwLike1 = ($h[2] -eq 0 -and $h[3] -eq 0 -and $len1 -ge 4 -and $len1 -le $fs.Length)

    if (-not $rdwLike1) { return 'FB or plain text (no RDW/BDW at start)' }

    # Peek next header to decide BDW vs RDW
    if ($fs.Length -ge 8) {
      [void]$fs.Seek(4, 'Begin')
      $h2 = New-Object byte[] 4
      [void]$fs.Read($h2,0,4)
      $len2 = ($h2[0] -shl 8) + $h2[1]
      $rdwLike2 = ($h2[2] -eq 0 -and $h2[3] -eq 0 -and $len2 -ge 4 -and $len2 -le $fs.Length)

      if ($rdwLike2) { return 'VB with BDW (block header) + RDW records' }
    }

    return 'VB (RDW records, no BDW)'
  }
  finally { $fs.Dispose() }
}

# ------- ls -la (ll) -------
function ll {
  param([string]$Path = ".")
  Get-ChildItem -LiteralPath $Path -Force |
    Select-Object Mode, LastWriteTime, Length, Name
}

# ------- sha256sum / sha1sum / md5sum / sha512sum -------
function sha256sum {
  param([Parameter(Mandatory=$true, ValueFromRemainingArguments=$true)][string[]]$Path)
  foreach ($p in $Path) {
    $h = (Get-FileHash -Algorithm SHA256 -LiteralPath $p).Hash.ToLower()
    "$h  $p"
  }
}

function sha1sum {

}

# ------- which -------
function which {
  param(
    [Parameter(Mandatory=$true)][string]$Name,
    [switch]$All
  )

  $cmds = Get-Command -Name $Name -ErrorAction SilentlyContinue
  if (-not $cmds) { return }

  if ($All) {
    foreach ($c in $cmds) {
      if ($c.Source) { $c.Source } elseif ($c.Path) { $c.Path }
    }
  } else {
    $c = $cmds | Select-Object -First 1
    if ($c.Source) { $c.Source } elseif ($c.Path) { $c.Path }
  }
}

# ------- whoami -------
function whoami {
  $id = [System.Security.Principal.WindowsIdentity]::GetCurrent()
  $id.Name
}

# ------- id (user + groups) -------
function id {
  $id = [System.Security.Principal.WindowsIdentity]::GetCurrent()
  Write-Output ("user  : {0}" -f $id.Name)
  Write-Output ("sid   : {0}" -f $id.User.Value)
  Write-Output "groups:"
  foreach ($g in $id.Groups) {
    try {
      $name = $g.Translate([System.Security.Principal.NTAccount]).Value
    } catch {
      $name = $g.Value
    }
    Write-Output ("  - {0}" -f $name)
  }
}

# ------- fgrep (fixed-string grep) -------
function fgrep {
  param(
    [Parameter(Mandatory=$true)][string]$Pattern,
    [string[]]$Path
  )

  if ($Path) {
    Select-String -Path $Path -Pattern $Pattern -SimpleMatch
  } else {
    $text = [Console]::In.ReadToEnd()
    $lines = $text -split "`r?`n"
    foreach ($line in $lines) {
      if ($line -like ("*" + $Pattern + "*")) { $line }
    }
  }
}

# ------- tail (+ follow) -------
function tail {
  param(
    [Parameter(Mandatory=$true)][string]$Path,
    [int]$Lines = 10,
    [switch]$Follow
  )

  if ($Follow) {
    Get-Content -LiteralPath $Path -Tail $Lines -Wait
  } else {
    Get-Content -LiteralPath $Path -Tail $Lines
  }
}

# ------- wc (pipeline or file path) -------
function wc {
  param([string[]]$Path)

  if ($Path) {
    foreach ($p in $Path) {
      $bytes = (Get-Item -LiteralPath $p).Length
      $lines = (Get-Content -LiteralPath $p -ReadCount 0).Length
      $text  = Get-Content -LiteralPath $p -Raw
      $words = ($text -split '\s+').Where({ $_ -ne '' }).Count
      "{0}`t{1}`t{2}`t{3}" -f $lines, $words, $bytes, $p
    }
    return
  }

  $inText = [Console]::In.ReadToEnd()
  $bytes2 = [Text.Encoding]::UTF8.GetByteCount($inText)
  $lines2 = ($inText -split "\r?\n").Where({ $_ -ne '' }).Count
  $words2 = ($inText -split '\s+').Where({ $_ -ne '' }).Count
  "{0}`t{1}`t{2}" -f $lines2, $words2, $bytes2
}

# ------- df (disk free) -------
function df {
  $drives = Get-PSDrive | Where-Object { $_.Provider -like "*FileSystem*" }
  foreach ($d in $drives) {
    $used  = [math]::Round(($d.Used  / 1GB), 2)
    $free  = [math]::Round(($d.Free  / 1GB), 2)
    $total = [math]::Round((($d.Used + $d.Free) / 1GB), 2)
    "{0}`tUsed:{1}GB`tFree:{2}GB`tTotal:{3}GB`t{4}" -f $d.Name, $used, $free, $total, $d.Root
  }
}

# ------- rm / mv / cp / touch -------
function rm {
  param([Parameter(Mandatory=$true)][string[]]$Path)
  Remove-Item -LiteralPath $Path -Recurse -Force
}

function mv {
  param(
    [Parameter(Mandatory=$true)][string]$Source,
    [Parameter(Mandatory=$true)][string]$Destination
  )
  Move-Item -LiteralPath $Source -Destination $Destination -Force
}

function cp {
  param(
    [Parameter(Mandatory=$true)][string]$Source,
    [Parameter(Mandatory=$true)][string]$Destination,
    [switch]$Recurse
  )
  if ($Recurse) {
    Copy-Item -LiteralPath $Source -Destination $Destination -Recurse -Force
  } else {
    Copy-Item -LiteralPath $Source -Destination $Destination -Force
  }
}

function touch {
  param([Parameter(Mandatory=$true)][string]$Path)

  if (Test-Path -LiteralPath $Path) {
    (Get-Item -LiteralPath $Path).LastWriteTime = Get-Date
  } else {
    New-Item -ItemType File -Path $Path | Out-Null
  }
}

# ------- find (simple: name pattern + type) -------
function find {
  param(
    [string]$Path = ".",
    [string]$Name = "*",
    [ValidateSet("*","f","d")][string]$Type = "*"
  )

  $items = Get-ChildItem -LiteralPath $Path -Recurse -Force -Filter $Name -ErrorAction SilentlyContinue

  if ($Type -eq "f") {
    $items = $items | Where-Object { -not $_.PSIsContainer }
  } elseif ($Type -eq "d") {
    $items = $items | Where-Object { $_.PSIsContainer }
  }

  foreach ($it in $items) { $it.FullName }
}

function cut {
  [CmdletBinding()]
  param(
    [Alias('b')]
    [Parameter(Mandatory = $true)]
    [string]$Bytes,                 # e.g. "11-", "27-31", "-23", "1,5,10-20"

    [Parameter()]
    [string]$Path                   # optional; if omitted, reads from stdin (raw bytes)
  )

  function Parse-ByteSpec {
    param([string]$Spec)
    $ranges = New-Object System.Collections.Generic.List[object]

    foreach ($tok in $Spec.Split(',')) {
      $t = $tok.Trim()
      if ($t -match '^\d+$') {
        $n = [int]$t
        $ranges.Add(@{ Start = $n; End = $n })
        continue
      }
      if ($t -match '^(?<a>\d+)?-(?<b>\d+)?$') {
        $a = if ($matches['a']) { [int]$matches['a'] } else { $null }
        $b = if ($matches['b']) { [int]$matches['b'] } else { $null }
        if ($a -eq $null -and $b -eq $null) {
          throw 'Invalid range: "-"'
        }
        $ranges.Add(@{ Start = $a; End = $b })
        continue
      }
      throw "Invalid byte range token: '$t'"
    }

    return ,$ranges
  }

  function Merge-Ranges {
    param([object[]]$RangesZeroBased)  # @{ S=int; E=int } inclusive

    $sorted = $RangesZeroBased | Sort-Object S, E
    $out = New-Object System.Collections.Generic.List[object]

    foreach ($r in $sorted) {
      if ($out.Count -eq 0) { [void]$out.Add(@{ S = $r.S; E = $r.E }); continue }
      $last = $out[$out.Count - 1]
      if ($r.S -le ($last.E + 1)) {
        $last.E = [Math]::Max($last.E, $r.E)
      } else {
        [void]$out.Add(@{ S = $r.S; E = $r.E })
      }
    }

    return ,$out
  }

  # Parse spec (1-based → clamp later per line)
  $spec = Parse-ByteSpec -Spec $Bytes

  # Read input bytes (file or stdin)
  if ($PSBoundParameters.ContainsKey('Path')) {
    $data = [IO.File]::ReadAllBytes($Path)
  } else {
    $inStream = [Console]::OpenStandardInput()
    $ms = New-Object IO.MemoryStream
    $inStream.CopyTo($ms)
    $data = $ms.ToArray()
    $ms.Dispose()
  }

  $outStream = [Console]::OpenStandardOutput()

  # Scan by lines (split on LF; preserve original newline: LF or CRLF)
  $i = 0
  while ($i -lt $data.Length) {
    $lineStart = $i
    while ($i -lt $data.Length -and $data[$i] -ne 0x0A) { $i++ }  # seek LF
    $lineEnd = $i                                 # points at LF or end
    $hasLF = ($i -lt $data.Length -and $data[$i] -eq 0x0A)
    if ($hasLF) { $i++ }                          # consume LF
    $hadCR = ($hasLF -and $lineEnd -gt $lineStart -and $data[$lineEnd - 1] -eq 0x0D)

    $contentEnd = if ($hadCR) { $lineEnd - 1 } else { $lineEnd }
    $lineLen = $contentEnd - $lineStart
    if ($lineLen -lt 0) { $lineLen = 0 }

    # Build zero-based ranges for THIS line, clamp to [0 .. lineLen-1]
    $z = New-Object System.Collections.Generic.List[object]
    foreach ($r in $spec) {
      $s = if ($r.Start -ne $null) { [int]$r.Start - 1 } else { 0 }
      $e = if ($r.End   -ne $null) { [int]$r.End   - 1 } else { $lineLen - 1 }

      if ($s -lt 0) { $s = 0 }
      if ($e -gt ($lineLen - 1)) { $e = $lineLen - 1 }
      if ($lineLen -gt 0 -and $s -le $e) {
        [void]$z.Add(@{ S = $s; E = $e })
      }
    }

    $merged = Merge-Ranges -RangesZeroBased $z.ToArray()

    # Emit selected bytes for this line
    foreach ($m in $merged) {
      $segLen = $m.E - $m.S + 1
      if ($segLen -gt 0) {
        $outStream.Write($data, $lineStart + $m.S, $segLen)
      }
    }

    # Re-emit the original newline
    if ($hasLF) {
      if ($hadCR) { $outStream.WriteByte(0x0D) }
      $outStream.WriteByte(0x0A)
    }
  }
}
